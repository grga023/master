-- ==========================================
-- KOCKA 1: cbProduktivnost
-- ==========================================

-- Upit 1: Produktivnost po ODELJENJIMA
SELECT 
  {[Measures].[Broj Sati], [Measures].[Trosak Rada], [Measures].[Satnica]} ON COLUMNS,
  [Odeljenje].[Odeljenje].Members ON ROWS
FROM [cbProduktivnost]

-- ==========================================
